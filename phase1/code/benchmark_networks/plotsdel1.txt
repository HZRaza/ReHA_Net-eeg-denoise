"""Plot validation CC and RRMSE vs epochs from checkpoint/state.pkl.
Usage: python plot_metric_epochs.py path/to/state.pkl [out_dir] [n_epochs=20]"""
import sys, os, pickle
import numpy as np
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt

path = sys.argv[1]
out = sys.argv[2] if len(sys.argv) > 2 else "."
n = int(sys.argv[3]) if len(sys.argv) > 3 else 20
s = pickle.load(open(path, "rb"))
if "val_cc_history" not in s or np.isnan(s["val_cc_history"]).all():
    sys.exit("This state.pkl has no CC/RRMSE history. Train with the new train_method.py first.")
cc = np.array(s["val_cc_history"], float)[:n]
rr = np.array(s["val_rrmse_history"], float)[:n]
if len(cc) < n:
    print(f"WARNING: only {len(cc)} epochs available; the title will say {len(cc)}.")
ep = np.arange(1, len(cc) + 1)
os.makedirs(out, exist_ok=True)
for y, name, label, color, fname in ((cc, "CC", "Correlation coefficient (CC)", "tab:blue", "cc_vs_epochs.png"),
                                     (rr, "RRMSE", "RRMSE", "tab:red", "rrmse_vs_epochs.png")):
    fig, ax = plt.subplots(figsize=(7, 4.5))
    ax.plot(ep, y, "o-", color=color, ms=4)
    ax.set_title(f"{name} across {len(ep)} epochs")
    ax.set_xlabel("Epoch"); ax.set_ylabel(label); ax.grid(alpha=.3)
    ax.set_xticks(ep[::2] if len(ep) > 10 else ep)
    fig.tight_layout(); fig.savefig(os.path.join(out, fname), dpi=200); plt.close(fig)
    print("saved", os.path.join(out, fname), f"(first {y[0]:.3f} -> last {y[-1]:.3f})")
